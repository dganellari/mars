cmake_minimum_required(VERSION 3.24)
project(mars_tree_layout_checks LANGUAGES CXX)
set(MARS_TREE_LANGUAGE "CUDA" CACHE STRING "CUDA or HIP")
if(NOT MARS_TREE_LANGUAGE MATCHES "^(CUDA|HIP)$")
    message(FATAL_ERROR "MARS_TREE_LANGUAGE must be CUDA or HIP")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_${MARS_TREE_LANGUAGE}_STANDARD 20)
enable_language(${MARS_TREE_LANGUAGE})
find_package(MPI REQUIRED COMPONENTS CXX)
if(MARS_TREE_LANGUAGE STREQUAL "HIP")
    # Imported targets must be visible where the test executables are linked.
    find_package(hip REQUIRED)
endif()
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(CSTONE_WITH_CUDA OFF CACHE BOOL "" FORCE)
set(CSTONE_WITH_HIP OFF CACHE BOOL "" FORCE)
set(CSTONE_WITH_${MARS_TREE_LANGUAGE} ON CACHE BOOL "" FORCE)
set(CSTONE_WITH_GPU_AWARE_MPI OFF CACHE BOOL "" FORCE)
if(NOT MARS_CSTONE_SOURCE_DIR)
    message(FATAL_ERROR "Pass the pinned MARS_CSTONE_SOURCE_DIR")
endif()
add_subdirectory("${MARS_CSTONE_SOURCE_DIR}" cornerstone)
get_filename_component(MARS_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(source "${CMAKE_CURRENT_SOURCE_DIR}/../tree_layout_gpu.cu")
set_source_files_properties("${source}" PROPERTIES LANGUAGE ${MARS_TREE_LANGUAGE})
foreach(mode IN ITEMS direct staged)
    add_executable(tree_${mode} "${source}")
    target_include_directories(tree_${mode} PRIVATE "${MARS_SOURCE_DIR}" "${MARS_CSTONE_SOURCE_DIR}/include")
    target_link_libraries(tree_${mode} PRIVATE cstone_gpu MPI::MPI_CXX cstone_thrust_incl)
    target_compile_definitions(tree_${mode} PRIVATE USE_CUDA)
    if(mode STREQUAL "direct")
        target_compile_definitions(tree_${mode} PRIVATE CSTONE_HAVE_GPU_AWARE_MPI)
    endif()
    if(MARS_TREE_LANGUAGE STREQUAL "CUDA")
        target_compile_options(tree_${mode} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--extended-lambda;--expt-relaxed-constexpr>)
        set_target_properties(tree_${mode} PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
    else()
        target_link_libraries(tree_${mode} PRIVATE hip::host)
    endif()
endforeach()

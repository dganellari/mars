cmake_minimum_required(VERSION 3.24)
project(mars_face_flux_checks LANGUAGES CXX)
enable_testing()
find_package(Mars REQUIRED)
find_package(MPI REQUIRED COMPONENTS CXX)
if(NOT TARGET Mars::mars_unstructured OR NOT MARS_GPU_LANGUAGE MATCHES "^(CUDA|HIP)$")
    message(FATAL_ERROR "Install the candidate GPU/Hypre unstructured MARS library first")
endif()
if(NOT DEFINED CMAKE_${MARS_GPU_LANGUAGE}_ARCHITECTURES)
    set(CMAKE_${MARS_GPU_LANGUAGE}_ARCHITECTURES "${MARS_GPU_ARCHITECTURES}")
endif()
set(CMAKE_${MARS_GPU_LANGUAGE}_STANDARD 20)
enable_language(${MARS_GPU_LANGUAGE})
set_source_files_properties(gpu.cu PROPERTIES LANGUAGE ${MARS_GPU_LANGUAGE})
add_executable(face_flux_gpu gpu.cu)
target_link_libraries(face_flux_gpu PRIVATE Mars::mars Mars::mars_unstructured MPI::MPI_CXX)
if(MARS_GPU_LANGUAGE STREQUAL "CUDA")
    set_target_properties(face_flux_gpu PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
endif()
foreach(ranks IN ITEMS 1 2 4)
    foreach(mode IN ITEMS tet_ddt_scs tet_ddt_source tet_k_scs tet_k_source
                          hex_ddt_scs hex_ddt_source hex_k_scs hex_k_source failed reject)
        add_test(NAME face_flux_${mode}_${ranks}
            COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} ${ranks} ${MPIEXEC_PREFLAGS}
            $<TARGET_FILE:face_flux_gpu> ${MPIEXEC_POSTFLAGS} ${mode})
        set_tests_properties(face_flux_${mode}_${ranks} PROPERTIES TIMEOUT 300 PROCESSORS ${ranks})
    endforeach()
endforeach()

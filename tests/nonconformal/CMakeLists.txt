cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(mars_nonconformal_checks LANGUAGES CXX)
    enable_testing()
endif()
add_executable(mars_nonconformal_flow_host flow.cpp)
target_compile_features(mars_nonconformal_flow_host PRIVATE cxx_std_20)
target_include_directories(mars_nonconformal_flow_host PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../..)
add_test(NAME mars_nonconformal_flow_host COMMAND mars_nonconformal_flow_host)
add_executable(mars_nonconformal_scalar_host scalar.cpp)
target_compile_features(mars_nonconformal_scalar_host PRIVATE cxx_std_20)
target_include_directories(mars_nonconformal_scalar_host PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../..)
add_test(NAME mars_nonconformal_scalar_host COMMAND mars_nonconformal_scalar_host)

if(MARS_ENABLE_MPI OR CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    find_package(MPI QUIET COMPONENTS CXX)
    if(MPI_CXX_FOUND)
        add_executable(mars_nonconformal_flow_accuracy_host flow_accuracy.cpp)
        target_compile_features(mars_nonconformal_flow_accuracy_host PRIVATE cxx_std_20)
        target_include_directories(mars_nonconformal_flow_accuracy_host PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../..)
        target_link_libraries(mars_nonconformal_flow_accuracy_host PRIVATE MPI::MPI_CXX)
        add_executable(mars_nonconformal_flow_fixture_host flow_channel.cpp)
        target_compile_features(mars_nonconformal_flow_fixture_host PRIVATE cxx_std_20)
        target_include_directories(mars_nonconformal_flow_fixture_host PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../..)
        target_link_libraries(mars_nonconformal_flow_fixture_host PRIVATE MPI::MPI_CXX)
        add_executable(mars_nonconformal_flow_channel_host ../../examples/distributed/unstructured/mars_nonconformal_flow.cpp)
        target_compile_features(mars_nonconformal_flow_channel_host PRIVATE cxx_std_20)
        target_include_directories(mars_nonconformal_flow_channel_host PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../..)
        target_link_libraries(mars_nonconformal_flow_channel_host PRIVATE MPI::MPI_CXX)
        add_executable(mars_nonconformal_mesh_host ../../examples/distributed/unstructured/mars_nonconformal_mesh.cpp)
        target_compile_features(mars_nonconformal_mesh_host PRIVATE cxx_std_20)
        target_include_directories(mars_nonconformal_mesh_host PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../..)
        target_link_libraries(mars_nonconformal_mesh_host PRIVATE MPI::MPI_CXX)
        add_executable(mars_nonconformal_channel_host channel.cpp)
        target_compile_features(mars_nonconformal_channel_host PRIVATE cxx_std_20)
        target_include_directories(mars_nonconformal_channel_host PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../..)
        target_link_libraries(mars_nonconformal_channel_host PRIVATE MPI::MPI_CXX)
        add_executable(mars_nonconformal_distributed_host distributed.cpp)
        target_compile_features(mars_nonconformal_distributed_host PRIVATE cxx_std_20)
        target_include_directories(mars_nonconformal_distributed_host PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../..)
        target_link_libraries(mars_nonconformal_distributed_host PRIVATE MPI::MPI_CXX)
        set(MARS_NONCONFORMAL_INPUT "" CACHE FILEPATH "Prepared Tet4 channel input for the mesh example checks")
        foreach(ranks 1 2 4)
            add_test(NAME mars_nonconformal_flow_fixture_host_${ranks}
                COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} ${ranks} ${MPIEXEC_PREFLAGS}
                        $<TARGET_FILE:mars_nonconformal_flow_fixture_host> ${MPIEXEC_POSTFLAGS})
            set_tests_properties(mars_nonconformal_flow_fixture_host_${ranks} PROPERTIES TIMEOUT 180)
            if(MARS_NONCONFORMAL_INPUT)
                add_test(NAME mars_nonconformal_mesh_host_${ranks}
                    COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} ${ranks} ${MPIEXEC_PREFLAGS}
                            $<TARGET_FILE:mars_nonconformal_mesh_host> ${MPIEXEC_POSTFLAGS}
                            --mesh ${MARS_NONCONFORMAL_INPUT} --check-linear-channel)
                set_tests_properties(mars_nonconformal_mesh_host_${ranks} PROPERTIES TIMEOUT 120)
            endif()
            add_test(NAME mars_nonconformal_channel_host_${ranks}
                COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} ${ranks} ${MPIEXEC_PREFLAGS}
                        $<TARGET_FILE:mars_nonconformal_channel_host> ${MPIEXEC_POSTFLAGS})
            set_tests_properties(mars_nonconformal_channel_host_${ranks} PROPERTIES TIMEOUT 60)
            add_test(NAME mars_nonconformal_distributed_host_${ranks}
                COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} ${ranks} ${MPIEXEC_PREFLAGS}
                        $<TARGET_FILE:mars_nonconformal_distributed_host> ${MPIEXEC_POSTFLAGS})
            set_tests_properties(mars_nonconformal_distributed_host_${ranks} PROPERTIES TIMEOUT 60)
        endforeach()
    endif()
endif()

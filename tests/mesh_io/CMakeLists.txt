cmake_minimum_required(VERSION 3.22)
project(mars_mesh_io_tests LANGUAGES CXX)
enable_testing()
find_package(MPI REQUIRED COMPONENTS CXX)
find_path(NETCDF_INCLUDE_DIR netcdf.h REQUIRED)
find_library(NETCDF_LIBRARY netcdf REQUIRED)
option(MARS_MESH_IO_CUDA "Compile the reader and deduplication with CUDA" OFF)
option(MARS_MESH_IO_HIP "Compile the reader and deduplication with HIP" OFF)
if(MARS_MESH_IO_CUDA AND MARS_MESH_IO_HIP)
    message(FATAL_ERROR "Choose one GPU backend")
endif()
if(MARS_MESH_IO_CUDA)
    enable_language(CUDA)
    set_source_files_properties(exodus_partition.cpp exodus_side_sets.cpp PROPERTIES LANGUAGE CUDA)
elseif(MARS_MESH_IO_HIP)
    enable_language(HIP)
    set_source_files_properties(exodus_partition.cpp exodus_side_sets.cpp PROPERTIES LANGUAGE HIP)
endif()
foreach(check exodus_partition exodus_side_sets)
    add_executable(mars_${check} ${check}.cpp)
    target_compile_features(mars_${check} PRIVATE cxx_std_20)
    set_target_properties(mars_${check} PROPERTIES CUDA_STANDARD 20 HIP_STANDARD 20)
    target_compile_definitions(mars_${check} PRIVATE MARS_HAVE_NETCDF OMPI_SKIP_MPICXX=1 MPICH_SKIP_MPICXX=1)
    target_include_directories(mars_${check} PRIVATE ${NETCDF_INCLUDE_DIR}
        ../../backend/distributed/unstructured/utils)
    target_link_libraries(mars_${check} PRIVATE MPI::MPI_CXX ${NETCDF_LIBRARY})
    foreach(ranks 1 2 4 8)
        add_test(NAME ${check}_${ranks}
            COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} ${ranks} ${MPIEXEC_PREFLAGS}
            $<TARGET_FILE:mars_${check}> ${MPIEXEC_POSTFLAGS} "${CMAKE_CURRENT_BINARY_DIR}/${check}-${ranks}")
        set_tests_properties(${check}_${ranks} PROPERTIES TIMEOUT 180 PROCESSORS ${ranks})
    endforeach()
endforeach()
if(CORNERSTONE_INCLUDE_DIRS)
    target_include_directories(mars_exodus_side_sets PRIVATE ${CORNERSTONE_INCLUDE_DIRS})
    target_compile_definitions(mars_exodus_side_sets PRIVATE MARS_MESH_IO_ROUTING)
endif()
add_executable(mars_exodus_no_netcdf no_netcdf.cpp)
target_compile_features(mars_exodus_no_netcdf PRIVATE cxx_std_20)
target_include_directories(mars_exodus_no_netcdf PRIVATE ../../backend/distributed/unstructured/utils)
add_test(NAME exodus_no_netcdf COMMAND mars_exodus_no_netcdf)

option(MARS_MESH_IO_DOMAIN "Check the installed GPU ElementDomain with public Hex8 boundaries" OFF)
if(MARS_MESH_IO_DOMAIN)
    if(NOT CORNERSTONE_INCLUDE_DIRS)
        message(FATAL_ERROR "Set CORNERSTONE_INCLUDE_DIRS to include the sparse boundary routing checks")
    endif()
    if(NOT MARS_MESH_IO_CUDA AND NOT MARS_MESH_IO_HIP)
        message(FATAL_ERROR "The ElementDomain gate requires CUDA or HIP")
    endif()
    find_package(Mars REQUIRED)
    if(MARS_MESH_IO_HIP)
        set_source_files_properties(exodus_domain.cpp PROPERTIES LANGUAGE HIP)
    else()
        set_source_files_properties(exodus_domain.cpp PROPERTIES LANGUAGE CUDA)
    endif()
    add_executable(mars_exodus_domain exodus_domain.cpp)
    target_compile_features(mars_exodus_domain PRIVATE cxx_std_20)
    set_target_properties(mars_exodus_domain PROPERTIES CUDA_STANDARD 20 HIP_STANDARD 20 CUDA_SEPARABLE_COMPILATION ON)
    target_include_directories(mars_exodus_domain PRIVATE ${NETCDF_INCLUDE_DIR})
    target_link_libraries(mars_exodus_domain PRIVATE Mars::mars Mars::mars_unstructured MPI::MPI_CXX ${NETCDF_LIBRARY})
    foreach(ranks 1 2 4 8)
        foreach(mode conformal blocks)
            add_test(NAME exodus_domain_${mode}_${ranks}
                COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} ${ranks} ${MPIEXEC_PREFLAGS}
                $<TARGET_FILE:mars_exodus_domain> ${MPIEXEC_POSTFLAGS}
                "${CMAKE_CURRENT_BINARY_DIR}/domain-${mode}-${ranks}" ${mode})
            set_tests_properties(exodus_domain_${mode}_${ranks} PROPERTIES TIMEOUT 180 PROCESSORS ${ranks})
        endforeach()
    endforeach()
endif()

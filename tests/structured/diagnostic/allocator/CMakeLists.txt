cmake_minimum_required(VERSION 3.24)
project(mars_mpi_allocator_probe LANGUAGES CXX CUDA)
find_package(MPI REQUIRED COMPONENTS CXX)
find_package(CUDAToolkit REQUIRED)
find_package(Kokkos 4.4 REQUIRED)
if(NOT Kokkos_ENABLE_CUDA)
    message(FATAL_ERROR "The allocator probe requires CUDA Kokkos")
endif()
add_executable(mars_mpi_allocator_probe main.cu)
target_compile_features(mars_mpi_allocator_probe PRIVATE cxx_std_20 cuda_std_20)
target_link_libraries(mars_mpi_allocator_probe PRIVATE Kokkos::kokkos MPI::MPI_CXX CUDA::cuda_driver)
if(Kokkos_ENABLE_CUDA_RELOCATABLE_DEVICE_CODE)
    set_target_properties(mars_mpi_allocator_probe PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
endif()

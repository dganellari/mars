cmake_minimum_required(VERSION 3.24)
project(mars_structured_diagnostic LANGUAGES CXX)
find_package(MPI REQUIRED COMPONENTS CXX)
if(NOT EXISTS "${KOKKOS_TOOLS_INCLUDE}/impl/Kokkos_Profiling_C_Interface.h")
    message(FATAL_ERROR "Use the pinned Kokkos source core/src directory")
endif()
add_library(mars_structured_trace SHARED trace.cpp)
target_compile_features(mars_structured_trace PRIVATE cxx_std_17)
target_include_directories(mars_structured_trace PRIVATE "${KOKKOS_TOOLS_INCLUDE}")
target_link_libraries(mars_structured_trace PRIVATE MPI::MPI_CXX)
if(MARS_DIAG_CUDA_INCLUDE)
    if(NOT EXISTS "${MARS_DIAG_CUDA_INCLUDE}/cuda_runtime_api.h")
        message(FATAL_ERROR "CUDA tracing requires the runtime headers")
    endif()
    target_include_directories(mars_structured_trace PRIVATE "${MARS_DIAG_CUDA_INCLUDE}")
    target_compile_definitions(mars_structured_trace PRIVATE MARS_DIAG_CUDA)
    target_link_libraries(mars_structured_trace PRIVATE ${CMAKE_DL_LIBS})
endif()
add_executable(mars_mpi_trace_probe probe.cpp)
target_link_libraries(mars_mpi_trace_probe PRIVATE MPI::MPI_CXX)
add_executable(mars_mpi_trace_linked_probe probe.cpp)
target_link_libraries(mars_mpi_trace_linked_probe PRIVATE mars_structured_trace MPI::MPI_CXX)

include(CTest)
if(BUILD_TESTING)
    add_executable(mars_mpi_trace_contract trace.cpp waitall_contract.cpp)
    target_compile_features(mars_mpi_trace_contract PRIVATE cxx_std_17)
    target_include_directories(mars_mpi_trace_contract PRIVATE "${KOKKOS_TOOLS_INCLUDE}")
    target_link_libraries(mars_mpi_trace_contract PRIVATE MPI::MPI_CXX)
    add_test(NAME trace_waitall_contract COMMAND mars_mpi_trace_contract)
endif()

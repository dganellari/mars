cmake_minimum_required(VERSION 3.24)
project(mars_structured_performance LANGUAGES CXX)
find_package(Mars REQUIRED)
if(MARS_GPU_LANGUAGE)
    enable_language(${MARS_GPU_LANGUAGE})
    set_source_files_properties(main.cpp PROPERTIES LANGUAGE ${MARS_GPU_LANGUAGE})
endif()
add_executable(mars_structured_performance main.cpp)
if(MARS_GPU_LANGUAGE STREQUAL "CUDA" AND Kokkos_ENABLE_CUDA_RELOCATABLE_DEVICE_CODE)
    set_target_properties(mars_structured_performance PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
endif()
target_link_libraries(mars_structured_performance PRIVATE Mars::mars)

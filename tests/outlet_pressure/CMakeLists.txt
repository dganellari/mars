cmake_minimum_required(VERSION 3.24)
project(mars_outlet_pressure_checks LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(MARS_SOURCE_DIR "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(solver "${MARS_SOURCE_DIR}/backend/distributed/unstructured/fem/mars_ns_pump_solver.hpp")
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/pressure_kernels.hpp"
    COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/extract.py" "${solver}"
        "${CMAKE_CURRENT_BINARY_DIR}/pressure_kernels.hpp"
    DEPENDS "${solver}" extract.py)
add_executable(outlet_pressure_algebra algebra.cpp "${CMAKE_CURRENT_BINARY_DIR}/pressure_kernels.hpp")
target_compile_features(outlet_pressure_algebra PRIVATE cxx_std_20)
target_include_directories(outlet_pressure_algebra PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME outlet_pressure_algebra COMMAND outlet_pressure_algebra)
option(MARS_PRESSURE_GPU_CHECKS "Check prescribed pressures with the installed GPU/Hypre MARS" OFF)
if(MARS_PRESSURE_GPU_CHECKS)
    find_package(Mars REQUIRED)
    find_package(MPI REQUIRED COMPONENTS CXX)
    if(NOT TARGET Mars::mars_unstructured OR NOT MARS_GPU_LANGUAGE MATCHES "^(CUDA|HIP)$")
        message(FATAL_ERROR "Install the candidate GPU/Hypre unstructured MARS library first")
    endif()
    if(NOT DEFINED CMAKE_${MARS_GPU_LANGUAGE}_ARCHITECTURES)
        set(CMAKE_${MARS_GPU_LANGUAGE}_ARCHITECTURES "${MARS_GPU_ARCHITECTURES}")
    endif()
    set(CMAKE_${MARS_GPU_LANGUAGE}_STANDARD 20)
    enable_language(${MARS_GPU_LANGUAGE})
    set_source_files_properties(gpu.cu PROPERTIES LANGUAGE ${MARS_GPU_LANGUAGE})
    add_executable(outlet_pressure_gpu gpu.cu)
    target_link_libraries(outlet_pressure_gpu PRIVATE Mars::mars Mars::mars_unstructured MPI::MPI_CXX)
    if(MARS_GPU_LANGUAGE STREQUAL "CUDA")
        set_target_properties(outlet_pressure_gpu PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
    endif()
    foreach(ranks IN ITEMS 1 2 4)
        foreach(mode IN ITEMS pressure reject refined)
            add_test(NAME outlet_${mode}_${ranks}
                COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} ${ranks} ${MPIEXEC_PREFLAGS}
                $<TARGET_FILE:outlet_pressure_gpu> ${MPIEXEC_POSTFLAGS} ${mode})
            set_tests_properties(outlet_${mode}_${ranks} PROPERTIES TIMEOUT 180 PROCESSORS ${ranks})
        endforeach()
    endforeach()
endif()

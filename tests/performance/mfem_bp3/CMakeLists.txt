cmake_minimum_required(VERSION 3.18)
project(mfem_bp3_driver LANGUAGES CXX)

find_package(MFEM 4.8 CONFIG REQUIRED)
find_package(MPI REQUIRED COMPONENTS CXX)
option(BP3_CPU_REFERENCE "Allow the CPU/MPI validation build" OFF)
if(MFEM_USE_CUDA OR NOT BP3_CPU_REFERENCE)
    find_package(CUDAToolkit REQUIRED)
endif()
if(NOT MFEM_USE_MPI OR (NOT MFEM_USE_CUDA AND NOT BP3_CPU_REFERENCE))
    message(FATAL_ERROR "The BP3 GPU baseline requires MFEM with MPI and CUDA")
endif()

add_executable(mfem_bp3 mfem_bp3.cpp)
target_compile_features(mfem_bp3 PRIVATE cxx_std_17)
# MFEM's exported host target omits CUDA headers needed by its public headers.
target_link_libraries(mfem_bp3 PRIVATE mfem MPI::MPI_CXX)
if(MFEM_USE_CUDA)
    target_link_libraries(mfem_bp3 PRIVATE CUDA::cudart CUDA::cusparse CUDA::cublas CUDA::curand)
endif()

option(BP3_PHASE_TESTS "Build the host phase-boundary checks" OFF)
if(BP3_PHASE_TESTS)
    enable_testing()
    add_executable(bp3_phase_test ../bp3_phases/test_phases.cpp)
    target_compile_features(bp3_phase_test PRIVATE cxx_std_17)
    target_link_libraries(bp3_phase_test PRIVATE MPI::MPI_CXX)
    foreach(mode off coarse detail unbalanced)
        add_test(NAME bp3_phase_${mode} COMMAND bp3_phase_test ${mode} ${CMAKE_CURRENT_BINARY_DIR}/phase-${mode})
    endforeach()
    set_tests_properties(bp3_phase_unbalanced PROPERTIES WILL_FAIL TRUE)
    add_executable(mfem_phase_wrappers_test test_phase_operators.cpp)
    target_compile_features(mfem_phase_wrappers_test PRIVATE cxx_std_17)
    target_link_libraries(mfem_phase_wrappers_test PRIVATE mfem MPI::MPI_CXX)
    if(MFEM_USE_CUDA)
        target_link_libraries(mfem_phase_wrappers_test PRIVATE CUDA::cudart CUDA::cusparse CUDA::cublas CUDA::curand)
    endif()
    add_test(NAME mfem_phase_wrappers COMMAND mfem_phase_wrappers_test)
    add_executable(mfem_h_transfer_test test_h_transfer.cpp)
    target_compile_features(mfem_h_transfer_test PRIVATE cxx_std_17)
    target_link_libraries(mfem_h_transfer_test PRIVATE mfem MPI::MPI_CXX)
    if(MFEM_USE_CUDA)
        target_link_libraries(mfem_h_transfer_test PRIVATE CUDA::cudart CUDA::cusparse CUDA::cublas CUDA::curand)
    endif()
    add_test(NAME mfem_h_transfer COMMAND mfem_h_transfer_test cpu)
endif()
